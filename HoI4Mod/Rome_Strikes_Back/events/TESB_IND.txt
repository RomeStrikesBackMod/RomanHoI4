add_namespace = tesb_india

# Start opening war
country_event = {
	id = tesb_india.1
	title = tesb_india.1.t
	desc = tesb_india.1.d
	picture = GFX_report_event_ger_civil_war

	is_triggered_only = yes
	option = {
		name = tesb_india.1.a
		IND = {
			add_state_claim = 611	# significant_number - Bhutan
			declare_war_on = {
				target = TIB
				type = annex_everything 
			}
		}
	}
}

# Invite country to faction
country_event = {
	id = tesb_india.10
	title = tesb_india.10.t
	desc = tesb_india.10.d
	picture = GFX_report_event_ger_civil_war

	is_triggered_only = yes
	# accept unconditionally
	option = {
		name = tesb_india.10.a
		IND = { country_event = { id = tesb_india.11 } }
		trigger = {
			NOT = { has_war_with = IND }
		}
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				NOT = { has_government = democratic }
			}
		}
	}
	# refuse
	option = {
		name = tesb_india.10.b
		IND = { country_event = { id = tesb_india.12 } }
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				has_government = democratic
			}
		}
	}
}

# country accepts
